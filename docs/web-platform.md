import {chromium} from 'playwright-core';
import assert from 'node:assert/strict';
const browser=await chromium.launch({channel:'msedge',headless:true});
try{const page=await browser.newPage();await page.route(/https:\/\/(?:[^/]*jsdelivr\.net|raw\.githubusercontent\.com)\//,route=>route.abort());await page.goto(process.env.XGHUB_TEST_ORIGIN||'http://127.0.0.1:1422');await page.getByRole('navigation',{name:'主导航'}).getByRole('link',{name:'漫画',exact:true}).click();await page.getByText('JavaScript 漫画插件',{exact:true}).click();await page.getByRole('button',{name:'同步内置仓库',exact:true}).click();await page.getByLabel('选择仓库漫画源').waitFor();const count=await page.getByLabel('选择仓库漫画源').locator('option').count();assert(count>0);await page.waitForFunction(()=>{const i=document.querySelector('img[alt="玄果漫画 X 书本图标"]');return i?.complete && i.naturalWidth>0});await page.setViewportSize({width:390,height:844});assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));console.log('PASS: same-site repository fetched with browser CDN/GitHub requests blocked, '+count+' entries; logo decoded; mobile width');}finally{await browser.close()}
